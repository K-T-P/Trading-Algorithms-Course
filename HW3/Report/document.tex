	\subsection{\lr{Bollinger Band}}
	\subsubsection{فرمول ریاضی}
	در ادامه اندیکاتور \lr{Bollinger Band} تعریف می‌شود. در این اندیکاتور، ما
	باند‌های بالایی،‌میانی و پایینی را تعریف و محاسبه می‌کنیم.
	
	باند بالایی از فرمول 
	$$SMA + Std \times Correlation$$
	و باند پایینی از فرمول
	$$SMA - Std \times Correlation$$
	محاسبه می‌شود. همچنین باند میانی با \lr{SMA} برابر است.
	\subsubsection{پیاده‌سازی}
	 حال برای محاسبه
	اندیکاتور \lr{Bollinger Band}، 
	ما ۴ تابع زیر را تعریف می‌کنیم:
	\begin{enumerate}
		\item \textbf{SMA}
		در این تابع، ما \lr{SMA} چند شمع قبلی را حساب می‌کنیم. تعداد این شمع‌ها توسط
		متغیر \lr{period} تعیین می‌شود.
		\item  \textbf{STD}
		در این تابع، ما انحراف معیار را محاسبه می‌کنیم.
		\item \textbf{UpperBound}
		در این تابع، ما با استفاده از توابع \lr{SMA} و \lr{STD} باند بالایی اندیکاتور
		را حساب می‌کنیم.
		\item \textbf{LowerBound}
		در این تابع، ما با استفاده از توابع \lr{SMA} و \lr{STD} باند پایینی را حساب
		می‌کنیم.
	\end{enumerate}
	در تصویر زیر، پیاده‌سازی اندیکاتور \lr{Bollinger Bond} نشان داده شده است.

	\subsection{\lr{Keltner Channel}}
	\subsubsection{فرمول ریاضی}
	حال اندیکاتور \lr{Keltner Channel} تعریف می‌شود. در این اندیکاتور، ما کانال های
	بالایی و پایینی را بصورت زیر تعریف می‌کنیم.
	\begin{eqnarray*}
		Upper Bound &=&SMA + ATR \times Correlation\\
		Lower Bound &=&SMA - ATR \times Correlation
	\end{eqnarray*}
	همچنین مقدار \lr{ATR} بصورت زیر بدست می‌آید.
	\begin{eqnarray*}
		ATR_{init} &=&\frac{\sum^{14}_{i=1} TR_{i}}{14}\\
		ATR_{j+1} &=&\frac{TR_{j} + (n-1)\times ATR_{j}}{n}
	\end{eqnarray*}
	که در اینجا منظور از \lr{n}، تعداد روزهای دوره است.
	\subsubsection{پیاده‌سازی}
	حال برای محاسبه باندهای بالایی و پایینی اندیکاتور \lr{Keltner Channel}، ما از
	توابع زیر استفاده می‌کنیم:
	\begin{enumerate}
		\item \textbf{\lr{SMA}} کارکردی مشابه تابع \lr{SMA} در اندیکاتور \lr{Bollinger
			Bound} دارد.
		\item \textbf{\lr{return\_max}} از میان مقادیر $High - Low$، $High - PrevClose$
		و $Low - PrevClose$ بیشترین مقدار را انتخاب می‌کند.
		\item \textbf{\lr{TR}}  به محاسبه مقادیر $High - Low$، $High - PrevClose$ و
		$Low - PrevClose$ می‌پردازد.
		\item \textbf{\lr{ATR}} با استفاده از تابع \lr{TR} به محاسبه \lr{ATR}
		می‌پردازد.
		\item \textbf{\lr{UpperBound}} با استفاده از مقادیر \lr{sma} و \lr{atr} به
		محاسبه باند بالایی می‌پردازد.
		\item \textbf{\lr{LowerBound}} با استفاده از مقادیر \lr{sma} و \lr{atr} به
		محاسبه باند پایینی می‌پردازد.
	\end{enumerate}
	حال در تصویر زیر، پیاده‌سازی این اندیکاتور نشان داده شده است.

	حال به تشریح تابع \lr{backtest} می‌پردازیم.
	
	در این تابع، ما مقادیر \lr{stock}, \lr{init\_fund} و \lr{fund\_rate} را ورودی

	سرمایه اولیه ما می‌باشد که با توجه به اشاره سوال به مقدار 1000، به طور پیشفرض
	برابر با 1000 می‌باشد. \lr{fund\_rate} نیز یک عدد در بازه صفر تا یک می‌باشد و
	اشاره دارد که در هر پوزیشن خرید یا فروش، چند درصد سرمایه را وارد کنیم.

	\end{figure}

	و \lr{RSI} را برروی قیمت رمزارز ورودی انجام می‌دهیم. سپس متغیرهای زیر را تعریف
	می‌کنیم:
	\begin{itemize}
		\item \textbf{\lr{oversold}} مقدار \lr{oversold} برای اندیکاتور \lr{RSI}
		\item \textbf{\lr{overbought}} مقدار \lr{overbought} برای اندیکاتور \lr{RSI}
		\item \textbf{\lr{current\_share}} تعداد سهم‌های خریداری شده

		\item \textbf{\lr{buy\_holding}} آیا پوزیشن \lr{buy} باز داریم یا خیر

	\subsection{تابع \lr{Backtest}}
	در این تابع، ما استراتژی خواسته شده در بخش دوم را پیاده‌سازی می‌کنیم. لازم به ذکر است که اندیکاتور \lr{Bollinger Band} در قسمت قبلی پیاده‌سازی شده است و از همان پیاده‌سازی استفاده می‌کنیم.
	\subsubsection{بخش ابتدایی تابع}
	\begin{figure}[H]
		\centering
		\includegraphics[scale=0.5]{pic_22}
		\caption{بخش ابتدایی تابع \lr{Backtest}}
	\end{figure}
	این تابع به عنوان ورودی، نمودار قیمت رمزارز، سرمایه اولیه و درصد ورود سرمایه را دریافت می‌کند که این موارد مشابه موارد متناظر در تابع \lr{backtest} قبلی است. همچنین تعدادی متغیر تعریف شده است که در زیر تشریح می‌شوند.
	\begin{itemize}
		\item \textbf{bollinger\_analysis} تحلیل اندیکاتور \lr{Bollinger Band} برروی نمودار قیمت رمزارز
		\item \textbf{current\_fund} سرمایه فعلی
		\item \textbf{current\_share} تعداد سهم‌های خریداری شده
		\item \textbf{sell\_holding} آیا پوزیشن \lr{sell} باز وجود دارد یا خیر
		\item \textbf{sell\_signal} سیگنال فروش
		\item \textbf{buy\_holding} آیا پوزیشن \lr{buy} باز وجود دارد یا خیر؟
		\item \textbf{buy\_signal} سیگنال خرید
		\item \textbf{sell\_price} قیمت سهم هنگام بازکردن پوزیشن فروش
		\item \textbf{events\_list} لیست زمان‌های بازکردن پوزیشن‌های خرید و فروش
	\end{itemize}
	\subsubsection{باز کردن پوزیشن \lr{Sell}}
	\begin{figure}[H]
		\centering
		\includegraphics[scale=0.5]{pic_23}
		\caption{بخش باز کردن پوزیشن فروش}
	\end{figure}
	در این بخش در شرط، ابتدا وجود سیگنال پوزیشن فروش و همچنین عدم وجود پوزیشن باز فروش را بررسی می‌کنیم. پس از ورود به بدنه شرط، سیگنال فروش (\lr{sell\_signal}) را برابر با \lr{False} قرار می‌دهیم. به این معنی که سیگنال فروش دریافت شده است. سپس در بخش شرط داخلی، بررسی می‌کنیم آیا پوزیشن \lr{buy} باز وجود دارد یا خیر. درصورت وجود، سهام‌ها را به قیمت روز به فروش می‌رسانیم. سپس متغیر \lr{sell\_holding} را برابر با \lr{True} قرار می‌دهیم تا بگوییم پوزیشن باز فروش وجود دارد. در ادامه، تعداد سهم‌های ممکن را با توجه به سرمایه نقدی موجود، درصد ورود سرمایه و قیمت هر سهم محاسبه می‌کنیم. سپس قیمت سهم‌ها را از سرمایه نقدی موجود کم می‌کنیم. همچنین قیمت هر سهم را نیز ذخیره می‌کنیم تا در هنگام بستن پوزیشن فروش، بتوانیم سود و زیان را نیز محاسبه کنیم.در آخر نیز تاریخ بازکردن پوزیشن به همراه نوع پوزیشن را ذخیره می‌کنیم.
