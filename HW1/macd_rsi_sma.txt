//@version=5
strategy("MACD Strategy", overlay=true)
fastLength = input(12)
slowlength = input(26)
MACDLength = input(9)
MACD = ta.sma(close, fastLength) - ta.sma(close, slowlength)
aMACD = ta.sma(MACD, MACDLength)
delta = MACD - aMACD

length = input( 14 )
overSold = input( 30 )
overBought = input( 70 )
price = close
vrsi = ta.rsi(price, length)
co = ta.crossover(vrsi, overSold)
cu = ta.crossunder(vrsi, overBought)

if (ta.crossover(delta, 0) or cu)
	strategy.entry("MacdLE", strategy.long, comment="MacdLE")
if (ta.crossunder(delta, 0) or co)
	strategy.entry("MacdSE", strategy.short, comment="MacdSE")
